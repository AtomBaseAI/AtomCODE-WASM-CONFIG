   141→async function api(path, { method = "GET", body, silent401 = false } = {}) {
   142→  const res = await app.api({ serverUrl: state.serverUrl, path, method, token: state.token, body });
   143→  if (res && res.status === 401 && !silent401 && state.token) {
   144→    clearSession();
   145→    showLogin("Session expired — please sign in again");
   146→    throw new Error(res.data?.error || "Unauthorized");
   147→  }
   148→  return res;
   149→}

   201→async function doLogin() {
   202→  const serverUrl = $("#login-server").value.trim().replace(/\/+$/, "");
   203→  const email = $("#login-email").value.trim();
   204→  const password = $("#login-password").value;
   205→  const errEl = $("#login-error");
   206→  if (!/^https?:\/\//i.test(serverUrl)) {
   207→    errEl.textContent = "Server address must start with http:// or https://";
   208→    errEl.classList.remove("hidden");
   209→    return;
   210→  }
   211→  state.serverUrl = serverUrl;
   212→  localStorage.setItem(LS.server, serverUrl);
   213→  const btn = $("#login-btn");
   214→  btn.disabled = true;
   215→  btn.innerHTML = `${icon("loader", "ic spin")} Signing in…`;
   216→  errEl.classList.add("hidden");
   217→  try {
   218→    const res = await api("/auth/login", { method: "POST", body: { email, password } });
   219→    if (!res.ok) {
   220→      errEl.textContent = res.data?.error || "Login failed";
   221→      errEl.classList.remove("hidden");
   222→      return;
   223→    }
   224→    state.token = res.data.token;
   225→    state.user = res.data.user;
   226→    localStorage.setItem(LS.token, state.token);
   227→    localStorage.setItem(LS.user, JSON.stringify(state.user));
   228→    toast(`Welcome back, ${state.user.name || state.user.email}!`);
   229→    showShell();
   230→  } catch (e) {
   231→    errEl.textContent = e.message || "Cannot reach the server";
   232→    errEl.classList.remove("hidden");
   233→  } finally {
   234→    btn.disabled = false;
   235→    btn.textContent = "Sign In";
   236→  }
   237→}

   298→  }).formatToParts(d);

   351→    const viewYear = Number(m.year);
